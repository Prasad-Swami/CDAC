query 1
select e.first_name, e.last_name, d.department_name
from employee e left join department d
ON e.department_id = d.department_id
where e.department_id is not null; 

query 2
select e.*, d.department_name from employee e
left join department d
ON e.department_id = d.department_id
where e.salary > 50000 AND d.department_name = 'IT'; 

query 3
select e.first_name, e.last_name, d.department_name
from employee e left join department d
on e.department_id = d.department_id
where first_name LIKE 'J%' AND email LIKE '%specialforce.com%';

query 4
select distinct department_name from department;

query 5
select sum(e.salary) as total_salary_expenditure , d.department_name
from employee e left join department d
on e.department_id = d.department_id
group by d.department_name;

query 6
select avg(e.salary) as avg_salary_finance, d.department_name
from employee e left join department d
on e.department_id = d.department_id
where department_name = 'Finance';

query 7
select min(e.salary) as min_salary, max(e.salary) as max_salary, d.department_name
from employee e left join department d
on e.department_id = d.department_id
where department_name = 'Sales';

query 8
select d.department_name, count(e.employee_id)
from employee e left join department d
on e.department_id = d.department_id
group by d.department_name;

query 9

query 10

query 11
select e.*, d.department_name
from employee e left join department d
ON e.department_id = d.department_id

query 12
select first_name, last_name, salary from employee
where salary between 30000 AND 70000
order by salary desc;
